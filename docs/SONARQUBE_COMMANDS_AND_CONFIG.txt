Environment Details:
- Operating System: Windows 11 (x64)
- JDK Version: Eclipse Adoptium OpenJDK 17 (17.0.16.8)
- Maven Version: Apache Maven 3.9.9
- SonarQube Version: SonarQube 9.9.8 LTS Community Edition
- SonarScanner Version: SonarScanner 5.0.1.3006
- SonarQube Server URL: http://localhost:9000
- Default Server Credentials: admin / admin


2. ALL TERMINAL COMMANDS USED
-------------------------------------------------------------------------------

[Environment Check Commands]
# Check available Java, javac, Docker, Maven, winget, choco
powershell -Command "Get-Command java, javac, mvn, docker, sonar-scanner -ErrorAction SilentlyContinue"
powershell -Command "java -version; javac -version"
powershell -Command "[Environment]::GetEnvironmentVariable('JAVA_HOME', 'Machine')"

[Download & Setup Commands]
# Download and extract Apache Maven 3.9.9
powershell -Command "Invoke-WebRequest -Uri 'https://archive.apache.org/dist/maven/maven-3/3.9.9/binaries/apache-maven-3.9.9-bin.zip' -OutFile 'maven.zip'; Expand-Archive -Path 'maven.zip' -DestinationPath '.'; Remove-Item 'maven.zip'"

# Download and extract SonarScanner CLI 5.0.1
powershell -Command "Invoke-WebRequest -Uri 'https://binaries.sonarsource.com/Distribution/sonar-scanner-cli/sonar-scanner-cli-5.0.1.3006-windows.zip' -OutFile 'sonar-scanner.zip'; Expand-Archive -Path 'sonar-scanner.zip' -DestinationPath '.'; Remove-Item 'sonar-scanner.zip'"

# Download and extract SonarQube 9.9.8 LTS Community Edition
powershell -Command "Invoke-WebRequest -Uri 'https://binaries.sonarsource.com/Distribution/sonarqube/sonarqube-9.9.8.100196.zip' -OutFile 'sonarqube.zip'; Expand-Archive -Path 'sonarqube.zip' -DestinationPath '.'; Remove-Item 'sonarqube.zip'"

[Compilation & Build Commands]
# Compile Java Application with Maven
.\apache-maven-3.9.9\bin\mvn.cmd compile

[SonarQube Server Startup Commands]
# Start SonarQube Server with JDK 17 Environment
start-sonar-jdk17.bat

# Check SonarQube Server Status API
powershell -Command "Invoke-RestMethod -Uri 'http://localhost:9000/api/system/status'"

[Static Code Analysis Execution Commands]
# Execute Static Analysis via Maven Sonar Plugin
.\apache-maven-3.9.9\bin\mvn.cmd sonar:sonar "-Dsonar.host.url=http://localhost:9000" "-Dsonar.login=admin" "-Dsonar.password=admin"

# Execute Static Analysis via SonarScanner CLI
.\sonar-scanner-5.0.1.3006-windows\bin\sonar-scanner.bat "-Dsonar.host.url=http://localhost:9000" "-Dsonar.login=admin" "-Dsonar.password=admin"

[REST API Results Query Commands]
# Fetch Issues and Measures via REST API
powershell -ExecutionPolicy Bypass -File "get_sonar_results.ps1"
